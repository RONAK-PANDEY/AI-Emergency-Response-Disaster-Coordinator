import { useState, useEffect } from "react";
import { X, AlertTriangle, Users, Clock, CheckCircle2 } from "lucide-react";

export interface Incident {
  id: string;
  type: string;
  severity: "low" | "medium" | "high" | "critical";
  description: string;
  peopleAffected: number;
  requiredTeams: string[];
  status: "pending" | "dispatched" | "in-progress" | "resolved";
}

interface IncidentDetailPanelProps {
  incident: Incident | null;
  isOpen: boolean;
  onClose: () => void;
  onDispatch?: (incidentId: string) => void;
}

const severityStyles: Record<Incident["severity"], string> = {
  low: "bg-emerald-500/15 text-emerald-400 border border-emerald-500/30",
  medium: "bg-amber-500/15 text-amber-400 border border-amber-500/30",
  high: "bg-orange-500/15 text-orange-400 border border-orange-500/30",
  critical: "bg-red-500/15 text-red-400 border border-red-500/30",
};

const statusStyles: Record<Incident["status"], string> = {
  pending: "bg-slate-500/15 text-slate-300 border border-slate-500/30",
  dispatched: "bg-blue-500/15 text-blue-400 border border-blue-500/30",
  "in-progress": "bg-violet-500/15 text-violet-400 border border-violet-500/30",
  resolved: "bg-emerald-500/15 text-emerald-400 border border-emerald-500/30",
};

export default function IncidentDetailPanel({
  incident,
  isOpen,
  onClose,
  onDispatch,
}: IncidentDetailPanelProps) {
  const [dispatching, setDispatching] = useState(false);

  useEffect(() => {
    setDispatching(false);
  }, [incident?.id]);

  if (!incident) return null;

  const handleDispatch = () => {
    setDispatching(true);
    onDispatch?.(incident.id);
    // simulate async action settling
    setTimeout(() => setDispatching(false), 900);
  };

  const isDispatched =
    incident.status === "dispatched" ||
    incident.status === "in-progress" ||
    incident.status === "resolved";

  return (
    <>
      {/* Backdrop */}
      <div
        className={`fixed inset-0 z-40 bg-black/60 backdrop-blur-sm transition-opacity duration-300 ${
          isOpen ? "opacity-100" : "pointer-events-none opacity-0"
        }`}
        onClick={onClose}
        aria-hidden="true"
      />

      {/* Panel */}
      <div
        role="dialog"
        aria-modal="true"
        aria-label="Incident details"
        className={`fixed right-0 top-0 z-50 h-full w-full max-w-md transform bg-slate-900 border-l border-slate-800 shadow-2xl transition-transform duration-300 ease-out ${
          isOpen ? "translate-x-0" : "translate-x-full"
        }`}
      >
        <div className="flex h-full flex-col">
          {/* Header */}
          <div className="flex items-start justify-between border-b border-slate-800 px-6 py-5">
            <div className="flex items-center gap-3">
              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-red-500/10 border border-red-500/20">
                <AlertTriangle className="h-5 w-5 text-red-400" />
              </div>
              <div>
                <h2 className="text-lg font-semibold text-slate-100">
                  {incident.type}
                </h2>
                <p className="text-xs text-slate-500">ID: {incident.id}</p>
              </div>
            </div>
            <button
              onClick={onClose}
              className="rounded-md p-1.5 text-slate-400 transition-colors hover:bg-slate-800 hover:text-slate-100"
              aria-label="Close panel"
            >
              <X className="h-5 w-5" />
            </button>
          </div>

          {/* Body */}
          <div className="flex-1 overflow-y-auto px-6 py-5 space-y-6">
            {/* Severity + Status badges */}
            <div className="flex flex-wrap gap-2">
              <span
                className={`inline-flex items-center rounded-full px-3 py-1 text-xs font-medium uppercase tracking-wide ${severityStyles[incident.severity]}`}
              >
                {incident.severity} severity
              </span>
              <span
                className={`inline-flex items-center gap-1.5 rounded-full px-3 py-1 text-xs font-medium uppercase tracking-wide ${statusStyles[incident.status]}`}
              >
                <Clock className="h-3 w-3" />
                {incident.status.replace("-", " ")}
              </span>
            </div>

            {/* Description */}
            <div>
              <h3 className="mb-2 text-xs font-semibold uppercase tracking-wide text-slate-500">
                Description
              </h3>
              <p className="text-sm leading-relaxed text-slate-300">
                {incident.description}
              </p>
            </div>

            {/* People affected */}
            <div>
              <h3 className="mb-2 text-xs font-semibold uppercase tracking-wide text-slate-500">
                People Affected
              </h3>
              <div className="flex items-center gap-2 rounded-lg border border-slate-800 bg-slate-800/40 px-4 py-3">
                <Users className="h-4 w-4 text-slate-400" />
                <span className="text-lg font-semibold text-slate-100">
                  {incident.peopleAffected.toLocaleString()}
                </span>
                <span className="text-sm text-slate-500">people</span>
              </div>
            </div>

            {/* Required teams */}
            <div>
              <h3 className="mb-2 text-xs font-semibold uppercase tracking-wide text-slate-500">
                Required Teams
              </h3>
              <div className="flex flex-wrap gap-2">
                {incident.requiredTeams.map((team) => (
                  <span
                    key={team}
                    className="rounded-md border border-slate-700 bg-slate-800 px-2.5 py-1 text-xs font-medium text-slate-300"
                  >
                    {team}
                  </span>
                ))}
              </div>
            </div>
          </div>

          {/* Footer */}
          <div className="border-t border-slate-800 px-6 py-4">
            <button
              onClick={handleDispatch}
              disabled={isDispatched || dispatching}
              className={`flex w-full items-center justify-center gap-2 rounded-lg px-4 py-2.5 text-sm font-semibold transition-colors ${
                isDispatched
                  ? "cursor-not-allowed bg-emerald-500/10 text-emerald-400 border border-emerald-500/30"
                  : dispatching
                  ? "cursor-wait bg-blue-600/70 text-white"
                  : "bg-blue-600 text-white hover:bg-blue-500"
              }`}
            >
              {isDispatched ? (
                <>
                  <CheckCircle2 className="h-4 w-4" />
                  Dispatched
                </>
              ) : dispatching ? (
                "Dispatching..."
              ) : (
                "Mark as Dispatched"
              )}
            </button>
          </div>
        </div>
      </div>
    </>
  );
}
