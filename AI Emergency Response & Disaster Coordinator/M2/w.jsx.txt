import 'leaflet/dist/leaflet.css';

import React, { useState } from 'react';
import { MapContainer, TileLayer, Marker, Popup } from 'react-leaflet';
import L from 'leaflet';

// Helper to create a simple colored pin icon using divIcon (no external image assets needed)
function createColorIcon(color) {
  return L.divIcon({
    className: 'custom-div-icon',
    html: `
      <div style="
        background-color: ${color};
        width: 24px;
        height: 24px;
        border-radius: 50% 50% 50% 0;
        transform: rotate(-45deg);
        border: 2px solid #ffffff;
        box-shadow: 0 0 4px rgba(0,0,0,0.5);
      "></div>
    `,
    iconSize: [24, 24],
    iconAnchor: [12, 24],
    popupAnchor: [0, -24],
  });
}

const fireIcon = createColorIcon('#e63946');   // red
const floodIcon = createColorIcon('#1d70ff');  // blue
const accidentIcon = createColorIcon('#f4a300'); // orange/yellow

// Severity → color mapping
const SEVERITY_COLORS = {
  critical: '#d90429',
  high: '#f4a300',
  medium: '#f2d600',
  low: '#2a9d8f',
};

function SeverityBadge({ severity }) {
  const color = SEVERITY_COLORS[severity] || '#999';
  return (
    <span
      style={{
        display: 'inline-block',
        backgroundColor: color,
        color: severity === 'medium' ? '#333' : '#fff',
        fontSize: '11px',
        fontWeight: 'bold',
        textTransform: 'uppercase',
        padding: '2px 8px',
        borderRadius: '10px',
        letterSpacing: '0.5px',
      }}
    >
      {severity}
    </span>
  );
}

function IncidentPopup({ incident, onViewDetails }) {
  const { icon, type, severity, description } = incident;
  return (
    <div style={{ minWidth: '180px' }}>
      <div style={{ marginBottom: '6px' }}>
        <strong>{icon} {type}</strong>
      </div>
      <div style={{ marginBottom: '6px' }}>
        <SeverityBadge severity={severity} />
      </div>
      <div style={{ marginBottom: '10px', fontSize: '13px', color: '#444' }}>
        {description}
      </div>
      <button
        onClick={() => onViewDetails(incident)}
        style={{
          background: '#1d70ff',
          color: '#fff',
          border: 'none',
          padding: '6px 12px',
          borderRadius: '4px',
          fontSize: '13px',
          cursor: 'pointer',
          width: '100%',
        }}
      >
        View Details
      </button>
    </div>
  );
}

const incidents = [
  {
    id: 1,
    position: [30.901, 75.857],
    icon: '🔥',
    type: 'Fire Reported',
    severity: 'critical',
    description: 'Near Ludhiana city center.',
    markerIcon: fireIcon,
  },
  {
    id: 2,
    position: [30.94, 75.83],
    icon: '🌊',
    type: 'Flood Alert',
    severity: 'medium',
    description: 'Low-lying area north of Ludhiana.',
    markerIcon: floodIcon,
  },
  {
    id: 3,
    position: [30.87, 75.90],
    icon: '🚧',
    type: 'Accident Reported',
    severity: 'high',
    description: 'Road incident southeast of Ludhiana.',
    markerIcon: accidentIcon,
  },
];

export default function PunjabMap() {
  const [selectedIncident, setSelectedIncident] = useState(null);

  const handleViewDetails = (incident) => {
    setSelectedIncident(incident);
  };

  return (
    <div style={{ position: 'relative', height: '100vh', width: '100vw' }}>
      <MapContainer
        center={[30.9, 75.8]}
        zoom={7}
        style={{ height: '100%', width: '100%' }}
      >
        <TileLayer
          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
          attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
        />

        {incidents.map((incident) => (
          <Marker key={incident.id} position={incident.position} icon={incident.markerIcon}>
            <Popup>
              <IncidentPopup incident={incident} onViewDetails={handleViewDetails} />
            </Popup>
          </Marker>
        ))}
      </MapContainer>

      {selectedIncident && (
        <div
          style={{
            position: 'absolute',
            top: 0,
            left: 0,
            right: 0,
            bottom: 0,
            background: 'rgba(0,0,0,0.5)',
            display: 'flex',
            alignItems: 'center',
            justifyContent: 'center',
            zIndex: 1000,
          }}
          onClick={() => setSelectedIncident(null)}
        >
          <div
            style={{
              background: '#fff',
              padding: '20px',
              borderRadius: '8px',
              width: '300px',
              maxWidth: '90%',
            }}
            onClick={(e) => e.stopPropagation()}
          >
            <h3 style={{ margin: '0 0 10px' }}>
              {selectedIncident.icon} {selectedIncident.type}
            </h3>
            <div style={{ marginBottom: '10px' }}>
              <SeverityBadge severity={selectedIncident.severity} />
            </div>
            <p style={{ fontSize: '14px', color: '#444' }}>{selectedIncident.description}</p>
            <p style={{ fontSize: '12px', color: '#888' }}>
              Coordinates: {selectedIncident.position[0]}, {selectedIncident.position[1]}
            </p>
            <button
              onClick={() => setSelectedIncident(null)}
              style={{
                marginTop: '10px',
                background: '#333',
                color: '#fff',
                border: 'none',
                padding: '6px 12px',
                borderRadius: '4px',
                cursor: 'pointer',
              }}
            >
              Close
            </button>
          </div>
        </div>
      )}
    </div>
  );
}
